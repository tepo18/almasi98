# ---------------- Helpers ----------------
def safe_b64decode(data: str) -> Optional[bytes]:
    if not isinstance(data, str):
        return None
    s = data.strip().replace("\n", "")
    s += "=" * ((4 - len(s) % 4) % 4)
    try:
        return base64.urlsafe_b64decode(s)
    except Exception:
        try:
            return base64.b64decode(s)
        except Exception:
            return None

def is_base64(s: str) -> bool:
    return safe_b64decode(s) is not None

def extract_host_from_link(link: str) -> Optional[str]:
    try:
        l = link.strip()
        if l.startswith("ss://"):
            main = l[5:].split("#")[0]
            decoded = safe_b64decode(main)
            if decoded:
                s = decoded.decode(errors="ignore")
                if "@" in s:
                    return s.rsplit("@",1)[1].split(":",1)[0]
            if "@" in main:
                return main.rsplit("@",1)[1].split(":",1)[0]
        if l.startswith("vmess://"):
            content = l.split("://",1)[1]
            maybe = safe_b64decode(content)
            if maybe:
                try:
                    jobj = json.loads(maybe.decode(errors="ignore"))
                    return jobj.get("add")
                except Exception:
                    pass
        m = re.search(r"@([^:/?#\s]+)", l)
        if m:
            return m.group(1)
        m2 = re.search(r"://([^:/?#\s]+)", l)
        if m2:
            return m2.group(1)
    except Exception:
        pass
    return None

def extract_protocol(link: str) -> str:
    if "://" in link:
        return link.split("://",1)[0].lower()
    return "unknown"

def classify_latency(ms: Optional[float]) -> Tuple[str,str]:
    if ms is None:
        return "bad","[bold red][BAD][/bold red]"
    try:
        val = float(ms)
    except Exception:
        return "bad","[bold red][BAD][/bold red]"
    if val < 150:
        return "good","[bold green][GOOD][/bold green]"
    if val < 300:
        return "warn","[bold yellow][WARN][/bold yellow]"
    return "bad","[bold red][BAD][/bold red]"

def ping_subprocess_avg(host: str, count: int = 3, timeout_s: int = 1) -> Optional[float]:
    try:
        output = subprocess.check_output(["ping","-c",str(count),"-W",str(timeout_s),host], stderr=subprocess.DEVNULL).decode(errors="ignore")
        m = re.search(r"rtt .* = [\d.]+/([\d.]+)", output)
        if m:
            return float(m.group(1))
    except Exception:
        return None
    return None

def ping_ping3_avg(host: str, count: int = PING3_COUNT, timeout: float = 1.0) -> Optional[float]:
    results = []
    for _ in range(count):
        try:
            r = ping(host, timeout=timeout, unit="ms")
            if r is not None:
                results.append(r)
        except Exception:
            pass
    if results:
        return sum(results)/len(results)
    return None

def parse_vless(link: str) -> Optional[Dict[str, Any]]:
    try:
        main = link.split("://", 1)[1]
        remark = ""
        if "#" in main:
            main, remark = main.split("#", 1)
            remark = urllib.parse.unquote(remark)
        if "?" in main:
            addr_port, params = main.split("?", 1)
            query = urllib.parse.parse_qs(params)
        else:
            addr_port = main
            query = {}
        if "@" in addr_port:
            user_id, host_port = addr_port.split("@", 1)
        else:
            user_id = ""
            host_port = addr_port
        if ":" in host_port:
            address, port = host_port.split(":", 1)
        else:
            address = host_port
            port = "443"
        return {
            "protocol": "vless",
            "raw": link,
            "address": address,
            "port": int(port) if str(port).isdigit() else 443,
            "id": user_id,
            "user": user_id,
            "params": {k: v[0] for k, v in query.items()},
            "remark": remark or link
        }
    except Exception:
        return None


def parse_vmess(link: str) -> Optional[Dict[str, Any]]:
    try:
        payload = link.split("://", 1)[1].strip()
        data = safe_b64decode(payload)
        if not data:
            return None
        jobj = json.loads(data.decode(errors="ignore"))
        address = jobj.get("add") or jobj.get("server")
        port = int(jobj.get("port") or 0) or 443
        uid = jobj.get("id") or jobj.get("uuid", "")
        network = jobj.get("net") or jobj.get("type") or "tcp"
        params = {"type": network}
        if jobj.get("tls"):
            params["security"] = "tls"
        if jobj.get("host"):
            params["host"] = jobj.get("host")
        if jobj.get("path"):
            params["path"] = jobj.get("path")
        remark = jobj.get("ps") or link
        return {
            "protocol": "vmess",
            "raw": link,
            "address": address,
            "port": port,
            "id": uid,
            "user": uid,
            "params": params,
            "remark": remark
        }
    except Exception:
        return None

def parse_generic(link: str) -> Tuple[Optional[str], Optional[Dict[str, Any]]]:
    link = link.strip()
    if not link:
        return None, None
    if link.startswith("vless://"):
        p = parse_vless(link)
        return ("vless", p) if p else (None, None)
    if link.startswith("vmess://"):
        p = parse_vmess(link)
        return ("vmess", p) if p else (None, None)
    if link.startswith("trojan://"):
        p = parse_trojan(link)
        return ("trojan", p) if p else (None, None)
    if link.startswith("ss://"):
        p = parse_ss(link)
        return ("ss", p) if p else (None, None)
    m = re.search(r"://([^:/?#\s]+)", link)
    host = m.group(1) if m else None
    if host:
        return ("unknown", {
            "protocol": "unknown",
            "raw": link,
            "address": host,
            "port": 443,
            "id": "",
            "user": "",
            "params": {},
            "remark": link
        })
    return None, None

# ---------------- Strip Protocol (for combo3) ----------------
def strip_protocol(link: str) -> str:
    for proto in PROTOCOL_PREFIXES:
        if link.startswith(proto):
            return link[len(proto):]
    return link

def generate_combo2(configs: list[str], cap: int = 50) -> list[str]:
    n = len(configs)
    if n < 2:
        return []
    indices = [(i,j) for i in range(n) for j in range(i+1,n)]
    random.shuffle(indices)
    combos=[]
    seen=set()
    for i,j in indices:
        combo=f"{configs[i]}+{configs[j]}"
        if combo not in seen:
            combos.append(combo)
            seen.add(combo)
        if len(combos) >= cap:
            break
    return combos

def generate_combo3(configs: list[str], cap: int = 50) -> list[str]:
    n = len(configs)
    if n < 3:
        return []
    cands=[]
    for a in range(n):
        for b in range(a+1,n):
            for c in range(b+1,n):
                cands.append((a,b,c))
                if len(cands) >= cap*8:
                    break
            if len(cands) >= cap*8:
                break
        if len(cands) >= cap*8:
            break
    random.shuffle(cands)
    combos=[]
    seen=set()
    for i,j,k in cands:
        A=configs[i]; B=configs[j]; C=configs[k]
        tail=strip_protocol(C)
        combo=f"{A}+{B}+ss//{tail}"
        if combo in seen:
            continue
        seen.add(combo)
        combos.append(combo)
        if len(combos) >= cap:
            break
    return combos

def collect_inputs() -> List[str]:
    print("[cyan]Input mode:[/cyan]")
    print("1) Paste configs manually")
    print("2) Fetch from subscription URLs (base64)")
    mode = Prompt.ask("Choose mode (1/2)", choices=["1", "2"], default="1")
    configs = []
    if mode == "2":
        import requests
        while True:
            url = input("Enter subscription URL (empty to finish): ").strip()
            if not url:
                break
            try:
                r = requests.get(url, timeout=12)
                r.raise_for_status()
                text = r.text.strip()
                payload = safe_b64decode(text)
                if payload:
                    payload = payload.decode(errors="ignore")
                else:
                    payload = text
                lines = [ln.strip() for ln in payload.splitlines() if ln.strip()]
                for ln in lines:
                    if any(ln.startswith(p) for p in PROTOCOL_PREFIXES) or is_base64(ln):
                        configs.append(ln)
            except Exception as e:
                print(f"[red]Failed {url}: {e}[/red]")
    else:
        print("[cyan]Paste configs line by line. Ctrl+D to end input.[/cyan]")
        while True:
            try:
                line = input()
                if line and line.strip():
                    configs.append(line.strip())
            except EOFError:
                break
    # remove duplicates
    unique = []
    seen = set()
    for c in configs:
        c2 = c.strip()
        if c2 and c2 not in seen:
            seen.add(c2)
            unique.append(c2)
    return unique

def process_pings_and_collect(configs: List[str]) -> Tuple[List[str], Dict[str, Tuple[str, Optional[float]]], Dict[str, int], List[Tuple[str, Dict[str, Any]]]]:
    ping_results = {}
    protocol_count = {}
    parsed_list = []

    def work(cfg: str):
        host = extract_host_from_link(cfg)
        sub = None
        p3 = None
        if host:
            sub = ping_subprocess_avg(host)
            p3 = ping_ping3_avg(host)
        status_sub, _ = classify_latency(sub)
        status_p3, _ = classify_latency(p3)
        avg = p3 if p3 is not None else sub
        overall = "bad"
        if status_p3 in ("good", "warn"):
            overall = status_p3
        elif status_sub in ("good", "warn"):
            overall = status_sub
        return cfg, host, overall, avg

    with Progress(SpinnerColumn(), "[progress.description]{task.description}", BarColumn(), "[progress.percentage]{task.percentage:>3.0f}%", TimeRemainingColumn()) as prog:
        task = prog.add_task("[cyan]Pinging hosts...", total=len(configs))
        with ThreadPoolExecutor(max_workers=min(MAX_THREADS, max(2, len(configs)))) as exc:
            futures = {exc.submit(work, c): c for c in configs}
            for fut in as_completed(futures):
                cfg, host, overall, avg = fut.result()
                ping_results[cfg] = (overall, avg)
                proto = extract_protocol(cfg)
                protocol_count[proto] = protocol_count.get(proto, 0) + 1
                prog.update(task, advance=1)
    for cfg in configs:
        _, parsed = parse_generic(cfg)
        if parsed:
            parsed_list.append((cfg, parsed))
    ordered = []
    for s in ("good", "warn", "bad"):
        ordered.extend([c for c, (st, _) in ping_results.items() if st == s])
    seen = set()
    ordered_unique = []
    for c in ordered:
        if c not in seen:
            ordered_unique.append(c)
            seen.add(c)
    return ordered_unique, ping_results, protocol_count, parsed_list

def ask_and_save_file(data: list[str] or list[Dict[str, Any]], default_name: str):
    fname = input(f"Enter output filename (default: {default_name}.txt): ").strip() or default_name
    if not (fname.lower().endswith(".txt") or fname.lower().endswith(".json")):
        if isinstance(data[0], dict):
            fname += ".json"
        else:
            fname += ".txt"
    outpath = os.path.join(OUTPUT_FOLDER, fname)
    mode = "w"
    encoding = "utf-8"
    with open(outpath, mode, encoding=encoding) as f:
        if isinstance(data[0], dict):
            json.dump(data, f, ensure_ascii=False, indent=2)
        else:
            for line in data:
                f.write(line + "\n")
    print(f"[green]✅ Saved {len(data)} items -> {outpath}[/green]")
