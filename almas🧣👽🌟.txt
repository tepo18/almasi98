#!/usr/bin/env python3
"""
Termux Multi Script Launcher - Python version
Equivalent of your bash menu, fixed and robust.
"""

import os
import subprocess
import shutil
import time
import sys

# Script files (same names as in your bash example)
scripts = {
    "1": ("Backup Python (.py)", "tepo.py"),
    "2": ("JSON Fragment (.py)", "tepo1.py"),
    "3": ("Ping Analyzer 1 (.py)", "tepo2.py"),
    "4": ("Ping Analyzer 2 (.py)", "tepo_new.py"),
    "5": ("Setup Termux (.sh)", "tepo.sh"),
    "6": ("Split Configs (.py)", "tepo4.py"),
}

# Colors (ANSI)
RED = "\033[31m"
GREEN = "\033[32m"
YELLOW = "\033[33m"
BLUE = "\033[34m"
MAGENTA = "\033[35m"
CYAN = "\033[36m"
WHITE = "\033[37m"
RESET = "\033[0m"

colors = [RED, GREEN, YELLOW, BLUE, MAGENTA, CYAN, WHITE]

def clear_screen():
    # works on Termux / Linux
    os.system("clear")

def pause(prompt="Press Enter to return to menu..."):
    try:
        input(prompt)
    except EOFError:
        # non-interactive fallback
        pass

def run_script(path):
    """
    Run a script file. Use python3 for .py and bash for .sh.
    Capture exit code and show a brief message.
    """
    if not os.path.isfile(path):
        print(f"{RED}File {path} not found!{RESET}")
        return

    # choose runner
    if path.endswith(".py"):
        runner = shutil.which("python3") or shutil.which("python") or "python3"
        cmd = [runner, path]
    elif path.endswith(".sh"):
        runner = shutil.which("bash") or "bash"
        cmd = [runner, path]
    else:
        # generic: try to execute directly
        cmd = [path]

    try:
        # run interactively so user sees output
        subprocess.run(cmd)
    except KeyboardInterrupt:
        print(f"\n{YELLOW}Script interrupted by user.{RESET}")
    except Exception as e:
        print(f"{RED}Failed to run {path}: {e}{RESET}")

def print_menu():
    clear_screen()
    print(f"{colors[0]}=============================================={RESET}")
    print(f"{colors[1]}         Termux Multi Script Launcher        {RESET}")
    print(f"{colors[2]}=============================================={RESET}")
    # Print items with colors similar to original
    for key in ["1","2","3","4"]:
        label, fname = scripts[key]
        color = colors[int(key) % len(colors)]
        print(f"{color}{key}) {label:<26} -> {fname}{RESET}")
    # 5 and 6 with MAGENTA/CYAN as in original
    print(f"{MAGENTA}5) {scripts['5'][0]:<26} -> {scripts['5'][1]}{RESET}")
    print(f"{CYAN}6) {scripts['6'][0]:<26} -> {scripts['6'][1]}{RESET}")
    print(f"{RED}7) Exit{RESET}")
    print(f"{colors[0]}----------------------------------------------{RESET}")

def main():
    # If running non-interactive, exit
    if not sys.stdin.isatty():
        print(f"{YELLOW}Non-interactive environment detected. Exiting.{RESET}")
        return

    while True:
        print_menu()
        try:
            choice = input("Enter your choice: ").strip()
        except EOFError:
            print()
            break

        if choice in scripts:
            _, script_file = scripts[choice]
            run_script(script_file)
            pause()
        elif choice == "7":
            print(f"{RED}Exiting launcher...{RESET}")
            break
        else:
            print(f"{RED}Invalid choice!{RESET}")
            time.sleep(1)

if __name__ == "__main__":
    main()
